cmake_minimum_required(VERSION 3.20)
project(CompilerLensLLVM VERSION 0.2.2 LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
set(COMPILERLENS_ZSTD_LIBRARY "" CACHE FILEPATH "zstd library for LLVM SDKs missing its CMake package")
if(COMPILERLENS_ZSTD_LIBRARY AND NOT TARGET zstd::libzstd_static)
  add_library(zstd::libzstd_static UNKNOWN IMPORTED)
  set_target_properties(zstd::libzstd_static PROPERTIES IMPORTED_LOCATION "${COMPILERLENS_ZSTD_LIBRARY}")
endif()
if(LLVM_VERSION_MAJOR LESS 22 OR LLVM_VERSION_MAJOR GREATER 23)
  message(FATAL_ERROR "CompilerLens supports LLVM 22/23 development toolchains")
endif()
list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
include(AddLLVM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS})
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/include")
add_definitions(${LLVM_DEFINITIONS})
add_library(CompilerLensLineage STATIC lib/Lineage.cpp)
llvm_update_compile_flags(CompilerLensLineage)
add_llvm_pass_plugin(CompilerLensPasses lib/Plugin.cpp)
target_link_libraries(CompilerLensPasses PRIVATE CompilerLensLineage)
add_executable(compilerlens-native tools/main.cpp)
llvm_update_compile_flags(compilerlens-native)
llvm_map_components_to_libnames(CLENS_LIBS Core IRReader Analysis Passes Support Object DebugInfoDWARF)
target_link_libraries(compilerlens-native PRIVATE CompilerLensLineage ${CLENS_LIBS})
set_target_properties(compilerlens-native PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS compilerlens-native RUNTIME DESTINATION compilerlens/_bin)
install(FILES ${LLVM_MAIN_INCLUDE_DIR}/../LICENSE.TXT DESTINATION compilerlens/_bin OPTIONAL)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(compilerlens-checkpoint-test tests/checkpoint.cpp)
  llvm_update_compile_flags(compilerlens-checkpoint-test)
  target_link_libraries(compilerlens-checkpoint-test PRIVATE CompilerLensLineage ${CLENS_LIBS})
  add_test(NAME lineage COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_reports.py
    $<TARGET_FILE:compilerlens-native> ${LLVM_TOOLS_BINARY_DIR}/opt $<TARGET_FILE:CompilerLensPasses>
    ${CMAKE_CURRENT_SOURCE_DIR}/tests $<TARGET_FILE:compilerlens-checkpoint-test> ${LLVM_TOOLS_BINARY_DIR}/llc)
endif()
